Lua API · OSTester
bus.write(address, data)
Записывает 16-битное значение data по 16-битному адресу address.
Возвращает: true при успехе, false при timeout.
local ok = bus.write(0x1000, 0x55AA)
if not ok then print("WRITE ERROR") end
bus.read(address)
Читает 16-битное значение с DATA[15:0] по адресу.
Успех: value. Ошибка: nil, "bus read timeout".
local value, err = bus.read(0x1000)
if value then
    print(string.format("DATA=0x%04X", value))
else
    print(err)
end
bus.write_ex(address, data)
Расширенная запись с подробным результатом операции.
Поля: ok, address, data, ready, irq, time_us, error.
local r = bus.write_ex(0x1000, 0x55AA)
print(r.ok, r.time_us, r.ready, r.irq)
if not r.ok then print(r.error) end
bus.read_ex(address)
Расширенное чтение.
Поля: ok, address, data, ready, irq, time_us, error.
local r = bus.read_ex(0x1000)
if r.ok then
    print(string.format("DATA=%04X", r.data))
else
    print(r.error)
end
bus.expect(address, expected [, mask])
Читает значение и сравнивает его с ожидаемым.
Поля: ok, matched, address, data, expected, mask, time_us, error.
local r = bus.expect(0x1000, 0x55AA)
if not r.ok then
    print(string.format("FAIL %04X got=%04X", r.address, r.data))
end

-- только младший байт:
local r2 = bus.expect(0x2000, 0x005A, 0x00FF)
bus.wait_ready(timeout_us)
Ждёт активного READY до заданного таймаута в микросекундах.
Возвращает true/false.
if not bus.wait_ready(5000) then
    print("READY timeout")
    return
end
bus.ready()
Возвращает текущее логическое состояние линии READY.
if bus.ready() then
    print("READY=1")
end
bus.irq()
Возвращает текущее состояние внешней линии IRQ.
if bus.irq() then
    print("IRQ active")
end
bus.timing(setup_us, pulse_us, hold_us)
Задаёт времена цикла записи.
-- setup=2 us, WR pulse=5 us, hold=3 us
bus.timing(2, 5, 3)
bus.invert_addr(enabled)
Включает/отключает физическую инверсию адресной шины. В Lua значения остаются логическими.
bus.invert_addr(false)
bus.invert_data(enabled)
Включает/отключает инверсию DATA[15:0].
bus.invert_data(false)
delay_us(us)
Микросекундная задержка внутри Lua-сценария.
delay_us(10)
print(...)
Выводит текст и значения в веб-журнал OSTester.
print("start")
print(string.format("DATA=%04X", value))
Пример: запись + проверка диапазона
bus.invert_addr(false)
bus.invert_data(false)
bus.timing(2, 5, 3)

local errors = 0
for addr = 0x0000, 0x00FE, 2 do
    local expected = 0x55AA
    local w = bus.write_ex(addr, expected)
    if not w.ok then
        errors = errors + 1
        print(string.format("WRITE ERROR %04X", addr))
    else
        local r = bus.expect(addr, expected)
        if not r.ok then
            errors = errors + 1
            print(string.format("DEFECT %04X expected=%04X actual=%04X", addr, expected, r.data))
        end
    end
end
print("Errors:", errors)
